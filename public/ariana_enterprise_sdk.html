Instalação rápida

const { ArianaEnterpriseClient } = require('./ariana-enterprise-sdk');

const ariana = new ArianaEnterpriseClient({
  apiKey: 'ari_live_xxxxx',
  environment: 'production'
});

await ariana.catalog.push([
  { sku:'ARI-0001', name:'Produto Teste', price:2299, stock:10 }
]);

await ariana.products.updateStock('ARI-0001', { stock: 8 });
await ariana.products.updatePrice('ARI-0001', { price: 2199 });
