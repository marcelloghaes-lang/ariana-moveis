  72: function buildDueDates(firstDueDate, count) {
  73:   const first = new Date(`${String(firstDueDate || '').slice(0, 10)}T12:00:00`);
  74:   if (Number.isNaN(first.getTime())) return [];
  75:   const day = first.getDate();
  76:   return Array.from({ length: count }, (_, index) => {
  77:     const d = new Date(first.getFullYear(), first.getMonth() + index, 1, 12, 0, 0, 0);
  78:     const lastDay = new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
  79:     d.setDate(Math.min(day, lastDay));
  80:     return d.toISOString().slice(0, 10);
  81:   });
